④ 遍历方法

forEach() — 遍历每个元素 无返回值

zhang_arr.forEach(function(item, index) {

});
map() — 映射为新数组 每个元素变个样

var zhang_doubled = zhang_arr.map(function(x) {
    return x * 2;
});
// → [2, 4, 6, 8]
