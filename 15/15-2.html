① 条件结构 (Conditionals)

if / else 判断温度 if · else
var zhang_temp = prompt_zjya();
var zhang_msg;
if (zhang_temp >= 35) {
    zhang_msg = '高温，注意防暑 ☀';
} else {
    zhang_msg = '温度正常 🌤';
}
document.write(zhang_msg);
if / else if / else 多段成绩评级 if · else if · else
var zhang_score = 85;
var zhang_grade;
if (zhang_score >= 90) {
    zhang_grade = '优秀 A';
} else if (zhang_score >= 80) {
    zhang_grade = '良好 B';
} else if (zhang_score >= 60) {
    zhang_grade = '及格 C';
} else {
    zhang_grade = '不及格 D';
}

② 循环结构 (Loops)

for 循环：输出 1~5 for
for (var i = 1; i <= 5; i++) {
    console.log(i);
}
while 循环：累加 1~100 while
var zhang_sum = 0, i = 1;
while (i <= 100) {
    zhang_sum += i;
    i++;
}
do…while 循环：至少执行一次 do · while
var zhang_c = 5;
do {
    console.log(zhang_c);
    zhang_c++;
} while (zhang_c < 3);
// 即使条件一开始就 false，循环体也执行一次
for…in 遍历对象属性 for…in
var zhang_info = { name: '张婧雅', class: '高级网页设计', score: 95 };
for (var key in zhang_info) {
    console.log(key + ' = ' + zhang_info[key]);
}
for…of 遍历数组元素 for…of
var zhang_fruits = ['🍎', '🍌', '🍇', '🍓'];
for (var f of zhang_fruits) {
    console.log(f);
}
